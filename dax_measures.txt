// ===== Power BI DAX measures =====
// Relationships (all many-to-one, single direction):
//   fact_sales[order_date]         -> dim_date[date]
//   fact_sales[customer_unique_id] -> dim_customers[customer_unique_id]
//   fact_sales[product_id]         -> dim_products[product_id]
//   rfm[customer_unique_id]        -> dim_customers[customer_unique_id]
// Mark dim_date as the Date table.

Total Revenue = SUM(fact_sales[revenue])
Total Orders = DISTINCTCOUNT(fact_sales[order_id])
Total Customers = DISTINCTCOUNT(fact_sales[customer_unique_id])
Avg Order Value = DIVIDE([Total Revenue], [Total Orders])

Revenue LY = CALCULATE([Total Revenue], SAMEPERIODLASTYEAR(dim_date[date]))
YoY Growth % = DIVIDE([Total Revenue] - [Revenue LY], [Revenue LY])

Revenue Prev Month = CALCULATE([Total Revenue], DATEADD(dim_date[date], -1, MONTH))
MoM Growth % = DIVIDE([Total Revenue] - [Revenue Prev Month], [Revenue Prev Month])

Revenue YTD = TOTALYTD([Total Revenue], dim_date[date])

Repeat Customers =
COUNTROWS(
    FILTER(
        VALUES(fact_sales[customer_unique_id]),
        CALCULATE(DISTINCTCOUNT(fact_sales[order_id])) > 1
    )
)
Repeat Customer % = DIVIDE([Repeat Customers], [Total Customers])

Avg Delivery Days = AVERAGE(fact_sales[delivery_days])
Late Delivery % = DIVIDE(SUM(fact_sales[is_late]), COUNTROWS(fact_sales))
Avg Review Score = AVERAGE(fact_sales[review_score])

Category Revenue Share % =
DIVIDE(
    [Total Revenue],
    CALCULATE([Total Revenue], ALL(dim_products[product_category]))
)

Category Rank =
RANKX(ALL(dim_products[product_category]), [Total Revenue], , DESC)

Champions Revenue % =
DIVIDE(
    CALCULATE([Total Revenue], rfm[Segment] = "Champions"),
    CALCULATE([Total Revenue], ALL(rfm[Segment]))
)
